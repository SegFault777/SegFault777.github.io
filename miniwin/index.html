# clone and build → build/os-image.img (1 MB)
$ git clone https://github.com/SegFault777/MiniWin.git
$ cd MiniWin && ./build.sh && tools/install_all.sh

# boot it in QEMU
$ qemu-system-i386 -drive file=build/os-image.img,format=raw

# with a NIC and a serial log (DHCP, DNS, TCP, TLS show up in serial.log)
$ qemu-system-i386 -drive file=build/os-image.img,format=raw \
    -netdev user,id=n0 -device rtl8139,netdev=n0 \
    -serial file:serial.log

# hear the PC-speaker beep (swap pa for dsound, coreaudio, sdl…)
$ qemu-system-i386 -drive file=build/os-image.img,format=raw \
    -audiodev pa,id=snd0 -machine pcspk-audiodev=snd0

MiniWin is running.
